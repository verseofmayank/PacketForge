cmake_minimum_required(VERSION 3.16)
project(PacketForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PACKETFORGE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "PacketForge currently targets Linux, including WSL2")
endif()
find_path(PCAP_INCLUDE_DIR pcap/pcap.h REQUIRED)
find_library(PCAP_LIBRARY pcap REQUIRED)
add_library(packetforge_core src/parsers/packet.cpp src/capture/pcap.cpp src/capture/live.cpp)
target_include_directories(packetforge_core PUBLIC include)
target_include_directories(packetforge_core SYSTEM PUBLIC third_party ${PCAP_INCLUDE_DIR})
target_link_libraries(packetforge_core PUBLIC ${PCAP_LIBRARY})
target_compile_options(packetforge_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(PACKETFORGE_SANITIZERS)
  target_compile_options(packetforge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(packetforge_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(packetforge src/main.cpp src/cli.cpp)
target_link_libraries(packetforge PRIVATE packetforge_core)
add_executable(sniffer src/sniffer.cpp src/cli.cpp)
target_link_libraries(sniffer PRIVATE packetforge_core)
add_executable(forwarder src/forwarder.cpp src/cli.cpp)
target_link_libraries(forwarder PRIVATE packetforge_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(packetforge_tests tests/unit.cpp)
  target_link_libraries(packetforge_tests PRIVATE packetforge_core)
  add_test(NAME unit COMMAND packetforge_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:packetforge> ${CMAKE_SOURCE_DIR})
endif()
install(TARGETS packetforge RUNTIME DESTINATION bin)
install(FILES config/rules.json DESTINATION share/packetforge)
